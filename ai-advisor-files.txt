59	                    .orElseThrow(() -> new ResourceNotFoundException("Conversation", request.getConversationId()));
60	            if (!conversation.getUser().getId().equals(userId))
61	                throw new ResourceNotFoundException("Conversation", request.getConversationId());
62	            messages = new ArrayList<>(conversation.getMessagesJson());
63	        } else {
64	            conversation = AiConversation.builder()
65	                    .user(user).messagesJson(new ArrayList<>()).build();

68	                    "content", buildSystemPrompt(user.getName(), buildProfileContext(userId))));
69	        }
70	
71	        messages.add(Map.of("role", "user", "content", request.getMessage()));
72	        String reply = callAiApi(messages);
73	        messages.add(Map.of("role", "assistant", "content", reply));
74	
75	        if (conversation.getTitle() == null) {
76	            String title = request.getMessage();
77	            conversation.setTitle(title.length() > 60 ? title.substring(0, 57) + "..." : title);
78	        }
79	        conversation.setMessagesJson(messages);
80	        conversation = conversationRepository.save(conversation);
81	
82	        return AiChatResponse.builder()

184	    private String callAiApi(List<Object> messages) {
185	        try {
186	            String apiUrl = aiProvider.equals("openai") ? openaiApiUrl : grokApiUrl;
187	            String apiKey = aiProvider.equals("openai") ? openaiApiKey : grokApiKey;
188	            String model  = aiProvider.equals("openai") ? "gpt-4o" : "llama-3.3-70b-versatile";
189	
190	            HttpHeaders headers = new HttpHeaders();
191	            headers.setContentType(MediaType.APPLICATION_JSON);
192	            headers.setBearerAuth(apiKey);
193	
194	            Map<String, Object> body = new HashMap<>();
195	            body.put("model", model);
196	            body.put("messages", messages);
197	            body.put("max_tokens", 2048);
198	            body.put("temperature", 0.7);
199	
200	            ResponseEntity<Map> response = restTemplate.postForEntity(
201	                    apiUrl, new HttpEntity<>(body, headers), Map.class);
202	
203	            if (response.getStatusCode() == HttpStatus.OK && response.getBody() != null) {
204	                List<Map<String, Object>> choices =
205	                        (List<Map<String, Object>>) response.getBody().get("choices");
206	                if (choices != null && !choices.isEmpty()) {
207	                    Map<String, Object> msg = (Map<String, Object>) choices.get(0).get("message");
208	                    return (String) msg.get("content");
209	                }
210	            }
211	            throw new BadRequestException("AI service returned empty response");
212	        } catch (BadRequestException e) {
213	            throw e;
214	        } catch (Exception e) {
215	            log.error("AI API call failed: {}", e.getMessage());
216	            throw new BadRequestException("AI service unavailable. Please try again shortly.");
217	        }
218	    }
